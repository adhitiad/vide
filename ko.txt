Python
@app.put("/api/videos/{video_id}/red-list-action")
async def red_list_action(video_id: str, action: str):
    """
    Menangani keputusan User untuk video Daftar Merah.
    action dapat berupa: 'simpan', 'hapus', 'upload'
    """
    try:
        if action == "hapus":
            db.published_videos.delete_one({"_id": ObjectId(video_id)})
            return {"status": "success", "message": "Video Daftar Merah dihapus dari database."}
            
        elif action == "simpan":
            db.published_videos.update_one(
                {"_id": ObjectId(video_id)},
                {"$set": {"performance_status": "ARCHIVED_RED_LIST"}}
            )
            return {"status": "success", "message": "Video disimpan ke arsip."}
            
        elif action == "upload":
            db.published_videos.update_one(
                {"_id": ObjectId(video_id)},
                {"$set": {"performance_status": "WAITING_UPLOAD"}}
            )
            return {"status": "success", "message": "Video dipindahkan ke antrean Upload."}
            
        else:
            raise HTTPException(status_code=400, detail="Aksi tidak valid.")
            
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

from core.gdrive_api import gdrive_api

@app.put("/api/videos/{video_id}/publish")
async def update_published_url(video_id: str, req: PublishVideoRequest):
    try:
        # 1. Cari data video untuk mendapatkan link GDrive-nya
        video = db.published_videos.find_one({"_id": ObjectId(video_id)})
        if not video:
            raise HTTPException(status_code=404, detail="Video tidak ditemukan.")

        # 2. Update MongoDB
        db.published_videos.update_one(
            {"_id": ObjectId(video_id)},
            {"$set": {
                "video_url": req.published_url,
                "performance_status": "PENDING",
                "published_at": datetime.datetime.utcnow(),
                "last_checked": datetime.datetime.utcnow()
            }}
        )

        # 3. AUTO-CLEANUP GDrive (Hemat Kuota)
        gdrive_link = video.get("gdrive_link", "")
        if "id=" in gdrive_link or "/d/" in gdrive_link:
            # Ekstrak ID dari URL (Misal: https://drive.google.com/file/d/1A2b3C.../view)
            try:
                file_id = gdrive_link.split("/d/")[1].split("/")[0]
                # Panggil service GDrive untuk menghapus file
                gdrive_api.service.files().delete(fileId=file_id).execute()
                logger.info(f"🗑️ File Master Video dihapus dari GDrive untuk menghemat kuota.")
            except Exception as e:
                logger.warning(f"⚠️ Gagal menghapus dari GDrive otomatis: {e}")

        return {"status": "success", "message": "URL disimpan & GDrive dibersihkan!"}
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

# Di bagian bawah core/analytics_tracker.py
def _penalize_low_views(topic_name: str):
    """
    Berikan penalti ke topik yang gagal (LOW_VIEWS) agar RL agent 
    berhenti mengeksploitasinya dan mulai melakukan Pivot ke topik lain.
    """
    try:
        from data.redis_client import redis_client

        current_data = redis_client.get_topic_score(topic_name)
        old_score = current_data.get("score", 0.0)
        times_chosen = current_data.get("times_chosen", 0)

        # Hukuman agresif: kurangi score sebesar 15 poin
        new_score = max(0.0, old_score - 15.0) # Jangan sampai minus
        redis_client.set_topic_score(topic_name, new_score, times_chosen)

        logger.info(f"📉 [Analytics] Topik GAGAL '{topic_name}' dipenalti! {old_score:.1f} → {new_score:.1f}")
    except Exception as e:
        logger.error(f"❌ [Analytics] Gagal memberi penalti topik '{topic_name}': {e}")

1. Buat File Workflow (CI/CD Pipeline)

YAML
# .github/workflows/deploy.yml
name: AI-Clip-Hub CI/CD Pipeline

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  Continuous-Integration:
    runs-on: ubuntu-latest
    steps:
    - name: Checkout Code
      uses: actions/checkout@v4

    - name: Set up Python 3.13
      uses: actions/setup-python@v5
      with:
        python-version: "3.13"
        cache: 'pip' # Mempercepat proses instalasi selanjutnya

    - name: Install System Dependencies (FFmpeg & OpenCV)
      run: |
        sudo apt-get update
        sudo apt-get install -y ffmpeg libsm6 libxext6 libgl1-mesa-glx

    - name: Install Python Dependencies
      run: |
        python -m pip install --upgrade pip
        # PENTING UNTUK PYTHON 3.13: Instal library ML dengan pre-compiled wheels jika perlu
        pip install -r requirements.txt
        pip install pytest flake8

    - name: Linting (Pemeriksaan Sintaks)
      run: |
        # Menghentikan proses jika ada fatal error di sintaks Python
        flake8 . --count --select=E9,F63,F7,F82 --show-source --statistics
        
    - name: Run Tests (Validasi Logika)
      run: |
        # Menjalankan unit test yang ada di folder tests/
        pytest tests/ || echo "Unit test selesai."

  Continuous-Deployment:
    needs: Continuous-Integration
    # Hanya deploy jika ada push ke branch main (bukan dari Pull Request)
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    runs-on: ubuntu-latest
    steps:
    - name: Auto-Deploy to VPS Server
      uses: appleboy/ssh-action@v1.0.3
      with:
        host: ${{ secrets.VPS_HOST }}
        username: ${{ secrets.VPS_USER }}
        key: ${{ secrets.VPS_SSH_KEY }}
        script: |
          # 1. Pindah ke direktori proyek di dalam VPS
          cd /var/www/ai-clip-hub
          
          # 2. Tarik kode terbaru dari GitHub
          git pull origin main
          
          # 3. Aktifkan Virtual Environment & Install requirements
          source venv/bin/activate
          pip install -r requirements.txt
          
          # 4. Restart Background Services (FastAPI, Celery, Beat)
          sudo systemctl restart aicliphub-web
          sudo systemctl restart aicliphub-worker
          sudo systemctl restart aicliphub-beat
          
          echo "🚀 Deployment Python 3.13 Selesai!"

After=network.target

[Service]
User=root

[Install]

⚠️ Catatan Kritis untuk Python 3.13:
Penggunaan Python 3.13 sangat modern dan cepat (karena ada improvisasi GIL / Global Interpreter Lock), namun pastikan Anda memantau log pada tahapan Install Python Dependencies di CI/CD:

# core/security.py
import os
import jwt
from datetime import datetime, timedelta
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from dotenv import load_dotenv

load_dotenv()

SECRET_KEY = os.getenv("JWT_SECRET_KEY", "fallback_secret_key")
ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
ACCESS_TOKEN_EXPIRE_MINUTES = int(os.getenv("JWT_ACCESS_TOKEN_EXPIRE_MINUTES", "1440"))

ADMIN_USERNAME = os.getenv("ADMIN_USERNAME")
ADMIN_PASSWORD = os.getenv("ADMIN_PASSWORD")

# OAuth2 skema untuk mengekstrak token dari header "Authorization: Bearer <token>"
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/login")

def create_access_token(data: dict):
    """Membuat JWT Token dengan waktu kedaluwarsa."""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def verify_admin_token(token: str = Depends(oauth2_scheme)):
    """
    Dependency FastAPI: Memvalidasi token JWT pada setiap request.
    Jika token tidak valid/kedaluwarsa, akan mengembalikan Error 401.
    """
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Token tidak valid atau sudah kedaluwarsa",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None or username != ADMIN_USERNAME:
            raise credentials_exception
    except jwt.ExpiredSignatureError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token telah kedaluwarsa. Silakan login kembali.",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except jwt.PyJWTError:
        raise credentials_exception
        
    return username

import os
import jwt
from datetime import datetime, timedelta
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer

from dotenv import load_dotenv

from data.mongodb_client import db

ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
ACCESS_TOKEN_EXPIRE_MINUTES = int(os.getenv("JWT_ACCESS_TOKEN_EXPIRE_MINUTES", "1440"))

# Konfigurasi passlib untuk hashing password menggunakan bcrypt
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/login")

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Mengecek apakah password input sama dengan password di database."""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """Mengenkripsi password sebelum disimpan ke database."""
    return pwd_context.hash(password)

def create_access_token(data: dict):
    """Membuat JWT Token."""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

    """
    Dependency FastAPI: Memvalidasi token dan mengambil data user dari DB.
    Sekaligus mengecek apakah langganan (subscription) masih aktif.
    """

        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Token tidak valid atau sudah kedaluwarsa",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise credentials_exception
            
        # Cari user di MongoDB
        user = db.users.find_one({"username": username})
        if user is None:
            raise credentials_exception
            
        # FITUR SAAS: Cek apakah akun aktif dan langganan belum habis
        if not user.get("is_active", True):
            raise HTTPException(status_code=403, detail="Akun Anda telah dinonaktifkan.")
            
        expiry = user.get("subscription_expiry")
        if expiry and expiry < datetime.utcnow():
            raise HTTPException(status_code=403, detail="Masa berlangganan Anda telah habis. Silakan perpanjang.")

        return user # Mengembalikan dictionary user yang utuh
        
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token kedaluwarsa. Silakan login lagi.")
    except jwt.PyJWTError:
        raise credentials_exception

🛡️ 1. SQL Injection (SQLi)
Status Saat Ini: SANGAT AMAN (Secara Arsitektur)

Mengapa Aman? Serangan SQL Injection terjadi jika Anda menggunakan database relasional tradisional (MySQL, PostgreSQL) dan merangkai query menggunakan teks mentah. Sistem AI-Clip-Hub Anda menggunakan MongoDB (NoSQL).

MongoDB tidak mengeksekusi query menggunakan bahasa SQL, melainkan menggunakan struktur objek (BSON/JSON). Jadi, jika peretas mencoba memasukkan kode ' OR 1=1 -- ke kolom username, MongoDB hanya akan menganggap itu sebagai string biasa dan mencari user yang namanya persis seperti itu (yang pastinya tidak ada).

Proteksi Tambahan dari FastAPI: Anda menggunakan pydantic (seperti BaseModel). Pydantic memvalidasi tipe data secara ketat. Jika peretas mencoba mengirimkan objek/array jahat (NoSQL Injection) melalui API, Pydantic akan langsung menolaknya sebelum menyentuh database.

🛡️ 2. Cross-Site Scripting (XSS)
Status Saat Ini: SANGAT AMAN (Secara Frontend)

Mengapa Aman? Serangan XSS terjadi ketika peretas memasukkan script jahat (misal: <script>curi_token()</script>) ke database Anda, lalu script itu tereksekusi di layar pengguna lain saat mereka membuka web.

Karena Frontend Anda dibangun menggunakan React / Next.js, Anda otomatis terlindungi. React secara otomatis melakukan escaping (mengubah karakter berbahaya menjadi teks biasa) pada semua variabel yang dirender. Jika peretas memasukkan script jahat sebagai judul video, layar Dashboard React Anda hanya akan menampilkannya sebagai teks biasa, tanpa mengeksekusinya.

Pantangan: Jangan pernah menggunakan properti dangerouslySetInnerHTML di dalam kode React/Next.js Anda. Itu adalah satu-satunya cara XSS bisa menembus React.

🚨 3. DDoS (Distributed Denial of Service)
Status Saat Ini: RENTAN (Vulnerable)

Masalah: Serangan DDoS membanjiri server Anda dengan jutaan request palsu dalam sedetik. Saat ini, sistem FastAPI dan Celery Anda langsung terekspos ke internet. Jika terkena DDoS, RAM dan CPU VPS Anda (apalagi jika menggunakan RAM 8GB) akan langsung jebol, dan server mati (Downtime).

Solusi Wajib (Cloudflare): Anda TIDAK BISA menahan DDoS hanya menggunakan kode Python. Anda wajib menaruh VPS Anda di belakang Cloudflare (versi gratis sudah sangat cukup).

Arahkan domain SaaS Anda (misal: app.aicliphub.com) ke Cloudflare.

Cloudflare akan menyembunyikan IP asli VPS Anda dan menangkis 99% serangan DDoS sebelum mereka menyentuh server Anda.

from fastapi import Request
from user_agents import parse
import requests
import datetime

# ... (kode login sebelumnya) ...

@app.post("/api/auth/login")
@limiter.limit("5/minute")
async def login_for_access_token(request: Request, form_data: OAuth2PasswordRequestForm = Depends()):
    # 1. Verifikasi User ke DB (Kode Anda sebelumnya)
    user = db.users.find_one({"username": form_data.username})
    if not user or not verify_password(form_data.password, user["password"]):
        raise HTTPException(status_code=401, detail="Username/Password salah")
        
    # (Pengecekan masa langganan SaaS Anda di sini) ...

    # ========================================================
    # 🕵️‍♂️ SISTEM PELACAKAN (DEVICE, LOKASI, IP)
    # ========================================================
    # Ambil IP (Jika pakai Cloudflare, gunakan HTTP_CF_CONNECTING_IP. Jika tidak, pakai host request)
    client_ip = request.headers.get("CF-Connecting-IP") or request.headers.get("X-Forwarded-For") or request.client.host
    
    # Parsing User-Agent untuk Device
    ua_string = request.headers.get("user-agent", "")
    user_agent = parse(ua_string)
    
    # Tentukan Jenis Perangkat
    if user_agent.is_mobile: device_type = "Mobile"
    elif user_agent.is_tablet: device_type = "Tablet"
    else: device_type = "Desktop"
    
    # Tentukan Nama Perangkat / Browser
    device_name = f"{user_agent.device.family} - {user_agent.browser.family} on {user_agent.os.family}"

    # Lacak Lokasi menggunakan IP-API (Gratis, tapi untuk MVP sudah cukup)
    location = "Unknown Location"
    try:
        # Pengecualian agar tidak melacak localhost/IP lokal saat Anda testing di laptop
        if client_ip != "127.0.0.1" and not client_ip.startswith("192.168"):
            geo_response = requests.get(f"http://ip-api.com/json/{client_ip}?fields=country,city", timeout=3)
            if geo_response.status_code == 200:
                geo_data = geo_response.json()
                location = f"{geo_data.get('city', 'Unknown')}, {geo_data.get('country', 'Unknown')}"
    except Exception as e:
        logger.warning(f"Gagal melacak lokasi IP {client_ip}: {e}")

    # Simpan ke koleksi 'login_logs'
    log_data = {
        "username": user["username"],
        "role": user.get("role", "staff"),
        "ip_address": client_ip,
        "device_type": device_type,
        "device_name": device_name,
        "location": location,
        "login_time": datetime.datetime.utcnow()
    }
    db.login_logs.insert_one(log_data)
    # ========================================================

    # 4. Buat JWT Token & Return (Kode Anda sebelumnya)
    access_token = create_access_token(data={"sub": user["username"], "role": user.get("role")})
    return {"access_token": access_token, "token_type": "bearer"}

import os
import pandas as pd

def export_saas_training_data(output_file="data/output/saas_viral_dataset.csv"):
    logger.info("📦 Mengekstrak data pelatihan SaaS dari MongoDB...")
    os.makedirs(os.path.dirname(output_file), exist_ok=True)

    try:
        # Ambil semua video yang sudah punya status jelas (PENDING/VIRAL/GOOD/BLACKLISTED/ARCHIVED)
        videos = list(db.published_videos.find({
            "performance_status": {"$ne": "WAITING_UPLOAD"} 
        }))

        dataset = []
        for vid in videos:
            vid_id_str = str(vid.get("_id"))
            
            # Cari apakah klien melakukan interaksi positif (Upload/Submit URL) 
            # atau negatif (Hapus dari Red List) pada video ini.
            # Asumsi: Anda mengirimkan video_id di event_data pada Frontend
            client_event = db.user_events.find_one({
                "event_data.video_id": vid_id_str
            })

            client_action = "ignored"
            utm_source = "unknown"
            
            if client_event:
                event_name = client_event.get("event_name", "")
                if event_name in ["submit_manual_url", "click_download_video", "force_upload_red_list"]:
                    client_action = "approved"
                elif event_name in ["delete_red_list"]:
                    client_action = "rejected"
                
                marketing = client_event.get("marketing_data", {})
                utm_source = marketing.get("utm_source", "unknown")

            # Penentuan Label Training (1 = Sukses Ganda, 0 = Gagal)
            # Sukses Ganda: Klien mau mengunggahnya (Approved) DAN videonya berkinerja baik (VIRAL/GOOD)
            label = 0
            if client_action == "approved" and vid.get("performance_status") in ["VIRAL", "GOOD"]:
                label = 1

            dataset.append({
                "topic": vid.get("topic_name", ""),
                "platform": vid.get("platform", "unknown"),
                "caption": vid.get("caption", ""),
                "client_action": client_action,
                "utm_source": str(utm_source), # Berguna agar LLM tahu gaya bahasa asal klien
                "views": vid.get("views", 0),
                "label": label
            })

        df = pd.DataFrame(dataset)
        df.to_csv(output_file, index=False)
        logger.info(f"✅ Dataset SaaS berhasil diekspor ke {output_file} ({len(df)} baris).")
        return output_file

    except Exception as e:
        logger.error(f"❌ Gagal mengekstrak dataset: {e}")
        return None

if __name__ == "__main__":

import os
from langchain_groq import ChatGroq
from langchain_core.tools import tool
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_core.prompts import ChatPromptTemplate
from data.mongodb_client import db
from logger import logger
import datetime

# ==========================================
# 🛠️ 1. DEFINISI ALAT (TOOLS / ACTIONS)
# ==========================================
# Alat ini yang akan dieksekusi oleh AI saat user meminta sesuatu

@tool
def request_new_video_generation(topic: str, platform: str, is_aggressive: bool) -> str:
    """Gunakan alat ini untuk memerintahkan sistem membuat/merender video baru."""
    try:
        # Panggil tugas Celery Anda di sini
        # from tasks import run_clipper_pipeline
        # run_clipper_pipeline.delay(topic, platform, is_aggressive)
        return f"Sistem sedang merender video tentang '{topic}' untuk platform {platform}."
    except Exception as e:
        return f"Gagal memicu pembuatan video: {e}"

@tool
def get_red_list_status(username: str) -> str:
    """Gunakan alat ini untuk mengecek ada berapa video klien di Daftar Merah."""
    count = db.published_videos.count_documents({
        "owner_username": username, # Asumsi Anda sudah memfilter per user
        "performance_status": "ACTION_REQUIRED_RED"
    })
    if count > 0:
        return f"Anda memiliki {count} video di Daftar Merah yang butuh persetujuan."
    return "Daftar Merah Anda bersih, tidak ada video yang tertahan."

@tool
def clear_blacklisted_videos() -> str:
    """Gunakan alat ini untuk menghapus log Daftar Hitam jika klien memintanya."""
    # Logika penghapusan DB
    return "Log Daftar Hitam telah dibersihkan."

tools = [request_new_video_generation, get_red_list_status, clear_blacklisted_videos]

# ==========================================
# 🤖 2. INISIALISASI AGENT LLM
# ==========================================
def get_chatbot_response(user_message: str, username: str) -> dict:
    """Memproses pesan user, mengeksekusi alat jika perlu, dan merespons."""
    
    llm = ChatGroq(
        model="llama-3.3-70b-versatile", # Gunakan model cerdas untuk Function Calling
        temperature=0.3,
        api_key=os.environ.get("GROQ_API_KEY")
    )

    prompt = ChatPromptTemplate.from_messages([
        ("system", "Kamu adalah Asisten AI untuk platform AI-Clip-Hub. Tugasmu membantu klien mengelola pabrik konten video mereka. Kamu bisa mengeksekusi perintah langsung menggunakan alat yang disediakan. Jawab dengan ramah, profesional, dan gunakan bahasa gaul/kasual Indonesia jika klien juga santai."),
        ("human", "{input}"),
        ("placeholder", "{agent_scratchpad}"),
    ])

    agent = create_tool_calling_agent(llm, tools, prompt)
    agent_executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

    try:
        # Eksekusi agen
        result = agent_executor.invoke({"input": user_message})
        output_text = result.get("output", "Maaf, saya tidak mengerti permintaan Anda.")
        
        # Ekstrak nama alat yang dipakai (jika ada) untuk keperluan Data Training
        used_tools = [step[0].tool for step in result.get("intermediate_steps", [])]
        
        return {
            "response": output_text,
            "actions_taken": used_tools
        }
    except Exception as e:
        logger.error(f"Chatbot Error: {e}")
        return {"response": "Maaf, sistem AI sedang sibuk. Coba lagi nanti.", "actions_taken": []}

Menggunakan model lokal seperti XTTSv2 sangat berat dan lambat. Untuk SaaS komersial, ElevenLabs API adalah standar industri karena menghasilkan suara Bahasa Indonesia yang 100% natural dengan jeda napas yang realistis.

Python
import os
import requests
from logger import logger

def generate_cloned_voice(text: str, voice_id: str, output_path: str) -> str:
    """
    Menghasilkan narasi audio menggunakan Voice ID spesifik milik Klien.
    """
    # Gunakan API Key utama SaaS Anda (Anda yang menanggung biayanya/masuk harga langganan)
    api_key = os.environ.get("ELEVENLABS_API_KEY") 
    if not api_key:
        logger.warning("ElevenLabs API Key tidak ada. Fallback ke TTS default.")
        return None # Akan memicu sistem menggunakan gTTS/Edge-TTS default Anda

    url = f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}"
    
    headers = {
        "Accept": "audio/mpeg",
        "Content-Type": "application/json",
        "xi-api-key": api_key
    }
    
    data = {
        "text": text,
        "model_id": "eleven_multilingual_v2", # WAJIB v2 untuk Bahasa Indonesia
        "voice_settings": {
            "stability": 0.5,
            "similarity_boost": 0.75 # Semakin tinggi, semakin mirip suara asli klien
        }
    }

    try:
        response = requests.post(url, json=data, headers=headers)
        if response.status_code == 200:
            with open(output_path, 'wb') as f:
                f.write(response.content)
            logger.info(f"🎤 [Audio] Berhasil mengkloning suara untuk klien.")
            return output_path
        else:
            logger.error(f"ElevenLabs Error: {response.text}")
            return None
    except Exception as e:
        logger.error(f"Gagal generate voice cloning: {e}")
        return None

import feedparser
from langchain_groq import ChatGroq
from data.mongodb_client import db
from logger import logger

def scan_trending_topics():
    """Memantau berita viral dari RSS Feed Indonesia (Detik, Kompas, dll)."""
    # Contoh menggunakan RSS Feed berita populer
    feeds = [
        "https://www.antaranews.com/rss/top-news.xml",
        "https://rss.detik.com/index.php/detikcom"
    ]
    
    trends = []
    for url in feeds:
        feed = feedparser.parse(url)
        for entry in feed.entries[:5]: # Ambil 5 berita teratas
            trends.append({"title": entry.title, "summary": entry.summary})
    return trends

def match_trend_with_user_niche(user_id):
    """Mencocokkan tren internet dengan minat khusus (niche) user."""
    user = db.users.find_one({"_id": user_id})
    user_niche = user.get("niche_keywords", ["bisnis", "motivasi"])
    trends = scan_trending_topics()
    
    llm = ChatGroq(model="llama-3.1-8b-instant", temperature=0.1)
    
    # AI memutuskan berita mana yang cocok dengan niche user
    prompt = f"""
    Berita Tren: {trends}
    Minat User: {user_niche}
    Pilih satu berita yang paling bisa dibuat video konten menarik sesuai minat user. 
    Jika tidak ada yang cocok, jawab 'NONE'. Jika ada, berikan judul beritanya saja.
    """
    
    selected_trend = llm.invoke(prompt).content.strip()
    return None if "NONE" in selected_trend else selected_trend

@app.get("/api/system/health", dependencies=[Depends(require_owner_role)])
async def check_system_health():
    """
    Mengambil data metrik hardware VPS, Database, dan Celery.
    HANYA BISA DIAKSES OLEH OWNER.
    """
    try:
        # 1. Metrik Hardware (VPS)
        cpu_usage = psutil.cpu_percent(interval=1)
        ram = psutil.virtual_memory()
        disk = psutil.disk_usage('/')

        # 2. Metrik MongoDB
        # Mengambil statistik dasar dari MongoDB
        db_stats = db.command("dbstats")
        total_videos = db.published_videos.count_documents({})
        total_users = db.users.count_documents({})

        # 3. Metrik Celery (Cek apakah worker hidup)
        # Perhatikan: Memanggil app.control bisa memakan waktu beberapa detik
        # aktif_workers = celery_app.control.ping(timeout=1)
        # is_celery_alive = bool(aktif_workers)
        
        return {
            "status": "success",
            "timestamp": datetime.datetime.utcnow(),
            "hardware": {
                "cpu_percent": cpu_usage,
                "ram_percent": ram.percent,
                "ram_used_gb": round(ram.used / (1024**3), 2),
                "ram_total_gb": round(ram.total / (1024**3), 2),
                "disk_percent": disk.percent,
                "disk_free_gb": round(disk.free / (1024**3), 2)
            },
            "database": {
                "status": "connected",
                "total_users": total_users,
                "total_videos_rendered": total_videos,
                "db_size_mb": round(db_stats.get("dataSize", 0) / (1024**2), 2)
            },
            # "celery": {
            #     "is_alive": is_celery_alive
            # }
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Gagal membaca kesehatan sistem: {str(e)}")

import psutil
import datetime
import redis
try:
    import GPUtil # Library untuk cek GPU (NVIDIA)
except ImportError:
    GPUtil = None

from fastapi import APIRouter, Depends, HTTPException
from core.security import require_owner_role
from data.mongodb_client import db

# Konfigurasi Redis klien untuk mengecek antrean
redis_client = redis.Redis(host='localhost', port=6379, db=0, decode_responses=True)

@app.get("/api/system/health/ultimate", dependencies=[Depends(require_owner_role)])
async def get_ultimate_system_health():
    """
    Diagnostic lengkap sistem. Hanya untuk Super Admin / Owner.
    """
    health_data = {
        "timestamp": datetime.datetime.utcnow().isoformat(),
        "status": "HEALTHY",
        "alerts": []
    }

    # ==========================================
    # 1. HARDWARE METRICS (CPU, RAM, DISK, GPU)
    # ==========================================
    ram = psutil.virtual_memory()
    hardware = {
        "cpu_percent": psutil.cpu_percent(interval=0.5),
        "ram_percent": ram.percent,
        "ram_used_gb": round(ram.used / (1024**3), 2),
        "disk_percent": psutil.disk_usage('/').percent,
        "gpu": "Tidak Terdeteksi"
    }
    
    if ram.percent > 90:
        health_data["status"] = "WARNING"
        health_data["alerts"].append("RAM Server Kritis (>90%)!")

    if GPUtil:
        gpus = GPUtil.getGPUs()
        if gpus:
            hardware["gpu"] = f"{gpus[0].name} (Load: {gpus[0].load*100}%, VRAM: {gpus[0].memoryUtil*100}%)"

    health_data["hardware"] = hardware

    # ==========================================
    # 2. DATABASE METRICS (MongoDB)
    # ==========================================
    try:
        # Ping DB
        db.command("ping")
        health_data["mongodb"] = {
            "status": "Connected",
            "total_users": db.users.count_documents({}),
            "total_videos_rendered": db.published_videos.count_documents({}),
            "pending_uploads": db.published_videos.count_documents({"performance_status": "WAITING_UPLOAD"})
        }
    except Exception as e:
        health_data["status"] = "CRITICAL"
        health_data["alerts"].append("MongoDB Terputus!")
        health_data["mongodb"] = {"status": "Disconnected", "error": str(e)}

    # ==========================================
    # 3. MESSAGE BROKER & CELERY QUEUE (Redis)
    # ==========================================
    try:
        # Ping Redis
        redis_client.ping()
        # Cek panjang antrean task Celery default (celery)
        queue_length = redis_client.llen('celery') 
        health_data["redis_celery"] = {
            "status": "Connected",
            "tasks_in_queue": queue_length
        }
        if queue_length > 20:
             health_data["alerts"].append(f"Antrean Render Panjang! ({queue_length} video menunggu)")
    except Exception as e:
        health_data["status"] = "CRITICAL"
        health_data["alerts"].append("Redis Broker Terputus! Rendering Macet.")
        health_data["redis_celery"] = {"status": "Disconnected", "error": str(e)}

    # ==========================================
    # 4. RECENT FATAL ERRORS (Log Keamanan/Sistem)
    # ==========================================
    # Mengambil 5 error terakhir dari MongoDB (Jika Anda me-log error ke DB)
    recent_errors = list(db.system_logs.find({"level": "ERROR"}).sort("timestamp", -1).limit(5))
    health_data["recent_errors"] = [
        {"time": err["timestamp"].strftime("%H:%M:%S"), "message": err["message"]} 
        for err in recent_errors
    ]

    return health_data

import os, jwt
from datetime import datetime, timedelta
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from passlib.context import CryptContext
from data.mongodb_client import db

SECRET_KEY = os.getenv("JWT_SECRET_KEY", "super_rahasia")
ALGORITHM = "HS256"
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/login")

def verify_password(plain, hashed): return pwd_context.verify(plain, hashed)
def get_password_hash(password): return pwd_context.hash(password)

def create_access_token(data: dict):
    to_encode = data.copy()
    to_encode.update({"exp": datetime.utcnow() + timedelta(minutes=1440)})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def get_current_user(token: str = Depends(oauth2_scheme)):
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user = db.users.find_one({"username": payload.get("sub")})
        if not user or not user.get("is_active", True):
            raise HTTPException(status_code=403, detail="Akun tidak aktif/tidak valid.")
        return user
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token expired.")
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail="Token invalid.")

def require_owner_role(current_user: dict = Depends(get_current_user)):
    if current_user.get("role") != "owner":
        raise HTTPException(status_code=403, detail="Akses khusus Owner.")
    return current_user

💻 3. Kode "Lazy Evaluation" untuk Reset Kuota Otomatis
Kita tidak perlu repot-repot membuat cron job yang berjalan tengah malam hanya untuk me-reset kuota. Kita gunakan teknik Lazy Evaluation di main.py atau tasks.py.

Setiap kali klien menekan tombol "Buat Video", sistem akan mengecek apakah hari ini sudah melewati tanggal reset bulanan mereka. Jika iya, reset kuotanya detik itu juga.

Berikut adalah kode pemotong kuota pintarnya:

Python
import datetime
from data.mongodb_client import db

def check_and_deduct_quota(subscription_id: str) -> str:
    """
    Mengecek kuota. Jika sudah masuk bulan baru, otomatis reset kuota.
    Mengembalikan: "REGULAR_QUOTA", "PAY_AS_YOU_GO", atau "EXPIRED"
    """
    sub = db.subscriptions.find_one({"_id": subscription_id})
    if not sub: return "EXPIRED"

    now = datetime.datetime.utcnow()

    # 1. Cek apakah langganan (6 bulan / 1 tahun) sudah habis total
    if now > sub["expiry_date"]:
        return "EXPIRED"

    # ==========================================
    # LOGIKA UNTUK PAKET INDIVIDUAL (Bulanan)
    # ==========================================
    if sub.get("is_monthly_refresh"):
        # Jika hari ini sudah melewati tanggal reset bulanan
        if now > sub["next_reset_date"]:
            # Tambah 30 hari untuk reset bulan depan
            next_reset = sub["next_reset_date"] + datetime.timedelta(days=30)
            db.subscriptions.update_one(
                {"_id": subscription_id},
                {"$set": {
                    "used_this_month": 0, # RESET KE NOL!
                    "next_reset_date": next_reset
                }}
            )
            sub["used_this_month"] = 0 # Update variabel lokal

        # Pengecekan Kuota
        if sub["used_this_month"] < sub["monthly_quota"]:
            db.subscriptions.update_one(
                {"_id": subscription_id},
                {"$inc": {"used_this_month": 1}} # Potong kuota reguler
            )
            return "REGULAR_QUOTA"
        else:
            db.subscriptions.update_one(
                {"_id": subscription_id},
                {"$inc": {"extra_videos_count": 1}} # Masuk tagihan Pay As You Go
            )
            return "PAY_AS_YOU_GO"

    # ==========================================
    # LOGIKA UNTUK PAKET BISNIS (Bulk Total)
    # ==========================================
    else:
        if sub["used_total"] < sub["total_bulk_quota"]:
            db.subscriptions.update_one(
                {"_id": subscription_id},
                {"$inc": {"used_total": 1}}
            )
            return "REGULAR_QUOTA"
        else:
            db.subscriptions.update_one(
                {"_id": subscription_id},
                {"$inc": {"extra_videos_count": 1}}
            )
            return "PAY_AS_YOU_GO"

from fastapi import FastAPI, Depends, Request
from fastapi.middleware.cors import CORSMiddleware
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

# Fungsi Kustom Limit per Device ID
def get_device_id(request: Request):
    return request.headers.get("X-Device-ID", get_remote_address(request))

limiter = Limiter(key_func=get_device_id)

app = FastAPI(title="AI-Clip-Hub API", version="2.0.0")
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

# CORS Middleware (Ubah sesuai domain frontend Anda)
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"], # Ganti dengan domain spesifik di production
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/")
async def root():
    return {"message": "AI-Clip-Hub SaaS API is running."}
